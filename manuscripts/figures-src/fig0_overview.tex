% Standalone figure: asr-gate method-overview schematic (paper's new Fig. 1).
% Two converging lanes -- offline calibration (top) and per-utterance
% test-time scoring (bottom) -- merging at the accept/defer decision, with
% the G1 certificate guarantee and the vacuous-at-target refusal case
% annotated as text callouts above/below the decision box (no extra boxes,
% per the figure-box budget: six pipeline boxes total).
% Every label reuses the paper's own notation/terms verbatim:
%   - decode artifacts, never the model (Sec. "Method: asr-gate", opening para)
%   - accept iff s >= lambda; select the most permissive lambda^* (Sec. G1)
%   - n_cal, speaker-disjoint calibration split (Sec. "Experimental setup")
%   - Learn-Then-Test, lambda-grid, EB p-values + Bonferroni-over-grid,
%     the frozen/adopted procedure (Sec. G1, "Disclosed pivot" paragraph)
%   - accepted-set macro-CER <= alpha w.p. >= 1-delta (Sec. G1)
%   - VACUOUS-AT-TARGET: no threshold certifies, nothing accepted, reported
%     not silently dropped (Sec. "The certificate holds on clean Aishell-1")
% No numeric values appear -- only the symbolic alpha, delta, n_cal, lambda,
% lambda^*, s -- per the figure brief. Drawn near-natural-size (~9cm wide) so
% that fonts stay >=8pt effective at TASLP columnwidth (~3.5in = 8.89cm).
% Compile: pdflatex fig0_overview.tex (see Makefile); output fig0_overview.pdf
% is copied unmodified into ../figures/ and ../taslp/figures/ (canonical and
% TASLP ports share one file).
\documentclass[border=1pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{tikz}
\usepackage{amsmath,amssymb}
\usetikzlibrary{arrows.meta,positioning,fit,calc}

\definecolor{cAccept}{HTML}{0072B2}
\definecolor{cDefer}{HTML}{D55E00}
\definecolor{cBoxFill}{HTML}{F2F2F2}
\definecolor{cBoxLine}{HTML}{404040}
\definecolor{cGuardFill}{HTML}{E8F1F8}

% TABLEFIX-2026-07-16: bumped 6.8/7.8 -> 7.5/8.5 (native 6.8pt scaled to the TASLP
%   columnwidth landed at ~6.9pt effective, under the 7pt floor; also removed decorative
%   \bfseries/\itshape below (box titles, "Guarantee:", accept/defer arrow labels, and the
%   vacuous-at-target callout) per the figure-typography directive -- structural distinction
%   (title vs. detail, accept vs. defer lane) is already carried by size and the existing
%   cAccept/cDefer colors, so the bold/italic was purely decorative.
\newcommand{\sub}[1]{{\fontsize{7.5}{8.5}\selectfont #1}}

\begin{document}
\begin{tikzpicture}[
  >=Latex,
  font=\fontsize{8}{9}\selectfont,
  pipebox/.style={draw=cBoxLine, thick, rounded corners=2pt, fill=cBoxFill,
    align=center, inner sep=3pt, minimum height=1.0cm},
  decbox/.style={draw=cBoxLine, very thick, rounded corners=2pt, fill=cBoxFill,
    align=center, inner sep=3pt},
]

% ---------- top lane: offline calibration ----------
\node[pipebox, text width=1.7cm] (calsplit) at (0,2.5)
  {Calibration split\\[2pt]\sub{$n_{\mathrm{cal}}$, speaker-disjoint}};
\node[pipebox, text width=1.95cm, right=0.22cm of calsplit] (ltt)
  {LTT\\certificate\\[2pt]\sub{$\lambda$-grid; EB $p$-values,\\Bonferroni $\Rightarrow \lambda^\star$}};

% ---------- bottom lane: per-utterance test time ----------
\node[pipebox, text width=1.55cm] (utt) at (0,0)
  {Utterance audio};
\node[pipebox, text width=1.75cm, right=0.22cm of utt] (backbone)
  {Frozen ASR backbone\\[2pt]\sub{decode artifacts\\(never the model)}};
\node[pipebox, text width=1.65cm, right=0.22cm of backbone] (score)
  {Confidence score $s$\\[2pt]\sub{accept iff $s\ge\lambda$}};

% ---------- merge: accept / defer decision ----------
% Anchored off score.east (always the rightmost lane edge, since the
% test-time lane has three boxes vs. the calibration lane's two) so the
% decision box never overlaps either lane; recentred to the mid-lane y.
\node[decbox, anchor=west, text width=1.55cm, minimum height=2.9cm]
  (dec) at ($(score.east)+(0.45,1.3)$)
  {Accept /\\defer\\[3pt]\sub{$s \ge \lambda^\star$?}};

% ---------- lane arrows ----------
\draw[->, thick] (calsplit) -- (ltt);
\draw[->, thick] (utt) -- (backbone);
\draw[->, thick] (backbone) -- (score);

% ---------- convergence arrows into the decision box ----------
\draw[->, thick] (ltt.east) -- ++(0.22,0) |- ($(dec.west)+(0,0.80)$);
\draw[->, thick] (score.east) -- ++(0.16,0) |- ($(dec.west)+(0,-0.80)$);

% ---------- decision outputs: stacked above/below dec (no extra width) ----------
% REVIEWFIX-D3-2026-07-16: in-figure text annotations set to black (were cAccept
%   "Guarantee:"/"accept" and cDefer "defer") per the annotation-color rail; the
%   box fills/borders and arrows keep their accept/defer colors (marks, not text).
\node[align=center, text width=2.05cm, above=0.42cm of dec,
      draw=cAccept, thick, rounded corners=1.5pt, fill=cGuardFill, inner sep=3pt]
  (guarantee) {\sub{Guarantee:}\\\sub{macro-CER $\le \alpha$ w.p.\ $\ge 1-\delta$}};
\draw[->, thick, cAccept] (dec.north) -- (guarantee.south);
\node[anchor=east, xshift=-1pt] at ($(dec.north)!0.5!(guarantee.south)+(-0.05,0)$) {\sub{accept}};

\node[align=center, text width=2.05cm, below=0.42cm of dec]
  (defer) {\sub{defer $\to$ human review}};
\draw[->, thick, cDefer, dashed] (dec.south) -- (defer.north);
\node[anchor=east, xshift=-1pt] at ($(dec.south)!0.5!(defer.north)+(-0.05,0)$) {\sub{defer}};

% ---------- vacuous-at-target callout, above the calibration lane ----------
% REVIEWFIX-D3-2026-07-16: reduced to a short label (was a full sentence inside the
%   figure); the "nothing accepted, reported not silently dropped" detail now lives
%   only in the caption (\cref{fig:overview}), per the in-figure-text rail.
\node[align=left, text width=3.0cm, above=0.42cm of ltt, xshift=-0.55cm]
  (vac) {\sub{if no $\lambda$ certifies:\\\textsc{vacuous-at-target}}};
\draw[-{Latex[length=1.5mm]}, thin, black!55, dash pattern=on 1.2pt off 1.2pt]
  (vac.south) -- (ltt.north);

\end{tikzpicture}
\end{document}
